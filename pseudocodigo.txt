algoritmo "CestaBasica"

var
   nome1, nome2, nome3: caractere
   situacao1, situacao2, situacao3: caractere
   anterior1, anterior2, anterior3: real
   atual1, atual2, atual3: real
   variacao1, variacao2, variacao3: real

inicio
   escreva("Produto 1: ")
   leia(nome1)
   escreva("Preço mês anterior: ")
   leia(anterior1)
   escreva("Preço mês atual: ")
   leia(atual1)

   escreva("Produto 2: ")
   leia(nome2)
   escreva("Preço mês anterior: ")
   leia(anterior2)
   escreva("Preço mês atual: ")
   leia(atual2)

   escreva("Produto 3: ")
   leia(nome3)
   escreva("Preço mês anterior: ")
   leia(anterior3)
   escreva("Preço mês atual: ")
   leia(atual3)

   se anterior1 <= 0 entao
      escreval(nome1, ": preço anterior inválido")
   senao
      variacao1 <- (atual1 - anterior1) / anterior1 * 100
      se variacao1 > 0 entao
         situacao1 <- "Aumento"
      senao
         se variacao1 < 0 entao
            situacao1 <- "Redução"
         senao
            situacao1 <- "Mantido"
         fimse
      fimse
      escreval(nome1, ": ", situacao1, " (", variacao1, "%)")
   fimse

   se anterior2 <= 0 entao
      escreval(nome2, ": preço anterior inválido")
   senao
      variacao2 <- (atual2 - anterior2) / anterior2 * 100
      se variacao2 > 0 entao
         situacao2 <- "Aumento"
      senao
         se variacao2 < 0 entao
            situacao2 <- "Redução"
         senao
            situacao2 <- "Mantido"
         fimse
      fimse
      escreval(nome2, ": ", situacao2, " (", variacao2, "%)")
   fimse

   se anterior3 <= 0 entao
      escreval(nome3, ": preço anterior inválido")
   senao
      variacao3 <- (atual3 - anterior3) / anterior3 * 100
      se variacao3 > 0 entao
         situacao3 <- "Aumento"
      senao
         se variacao3 < 0 entao
            situacao3 <- "Redução"
         senao
            situacao3 <- "Mantido"
         fimse
      fimse
      escreval(nome3, ": ", situacao3, " (", variacao3, "%)")
   fimse
